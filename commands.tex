\makeatletter
\newcommand{\pdv}[3][1]{\ensuremath{%
    \ifnum#1=1
        \def\@pdv@exp{}
    \else
        \def\@pdv@exp{^{#1}}
    \fi
    \frac{\partial{\@pdv@exp} #2}{\partial #3{\@pdv@exp}}}%
}

\newcommand{\dv}[3][1]{\ensuremath{%
    \ifnum#1=1
        \def\@dv@exp{}
    \else
        \def\@dv@exp{#1}
    \fi
    \frac{{\mathrm{d}}^{\@dv@exp} #2}{{\mathrm{d}} #3^{\@dv@exp}}}%
}

\newcommand{\dd}[2][1]{\ensuremath{%
    \ifnum#1=1
        \def\@dd@exp{}
    \else
        \def\@dd@exp{#1}
    \fi
    {\mathrm{d}}^{\@dd@exp} #2}%
}
\makeatother
