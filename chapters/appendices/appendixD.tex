The possible mass loss prescriptions (IML1/IML2) are:

\begin{enumerate}
  \setcounter{enumi}{-1}
  \item No mass loss
  \item \citet{Reimers:1975}
  \item \citet{Bloecker:1995}
  \item \citet{Vassiliadis:1993}
  \item \citet{De_jager:1988, Nugis:2000, Vink:2001} with interpolation errors described in \citet{Richards:2024}
  \item \citet{De_jager:1988, Nugis:2000, Vink:2001} \citep[described in][]{Eldridge:2017}
  \item \citet{De_jager:1988, Nugis:2000, Vink:2001} with slowly turning on \gls{wr} rates described in \citet{Richards:2024}
  \item \citet{De_jager:1988, Nugis:2000, Vink:2001} with no \gls{wr} rates, described in \citet{Richards:2024}
  \item Unimplemented
  \item \gls{zams} evolution to a specific \gls{zams} mass, custom routine by Jan Eldridge
\end{enumerate}
