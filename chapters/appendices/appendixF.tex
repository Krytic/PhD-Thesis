\section{Roche lobe formula}

We follow the derivation given in \citet{Rose:1998} to find an expression for the Roche potential. First we note that in a rotating reference frame in Cartesian coordinates, we have

\begin{equation}
    \overbrace{\frac{GM_1 M_2}{a^2}\vu{x}}^{\mathclap{\text{gravitational force}}}=\underbrace{-M_1 \vec{\omega}\times\pqty{\vec{\omega}\times\vec{r}_1}}_\text{centrifugal force},
\end{equation}

where $\vu{x}$ is the line between the center of mass of both stars, $a$ the separation, $\vec{\omega}$ the angular velocity, and $\vec{r}_1$ the position vector of the primary star. The x-ordinates of the centres of mass are given by

\begin{align}
    x_1 &= \frac{M_2}{M_1+M_2}a \\
    x_2 &= \frac{-M_1}{M_1+M_2}a
\end{align}

Since we are considering a rotating coordinate system we have that the centrifugal force is given by

\begin{equation}
    -M_1 \vec{\omega}\times\pqty{\vec{\omega}\times\vec{r}_1} = M_1\omega^2 x_1,
\end{equation}

and hence

\begin{align}
    \frac{GM_1 M_2}{a^2}\vu{x} &= M_1\omega^2 x_1 \notag\\
    \omega^2 &= \frac{GM_1 M_2}{M_1 x_1 a^2}\vu{x} \notag\\
    &= \frac{GM_2}{x_1 a^2}\vu{x} \notag\\
    &= \frac{GM_2\pqty{M_1+M_2}}{M_2 \cdot a\cdot a^2}\vu{x} \notag\\
    &= \frac{G\pqty{M_1+M_2}}{a^3}\vu{x}.
\end{align}

The centrifugal potential of a test mass $m$ at $(x,y)$ is hence

\begin{equation}
    V = -\frac12 m\omega^2\pqty{x^2+y^2} = -\frac{Gm\pqty{M_1+M_2}\pqty{x^2+y^2}}{2a^3},
\end{equation}

and thus its energy is

\begin{equation}
    E = \frac12 m\pqty{\dot{x}^2+\dot{y}^2} + V(x,y),
\end{equation}

where

\begin{equation}
    V(x,y) = \overbrace{-\frac{GmM_1}{\sqrt{\pqty{x-x_1}^2+y^2}}}^\text{G.P.E. due to primary}-\overbrace{\frac{GmM_2}{\sqrt{\pqty{x-x_2}^2+y^2}}}^\text{G.P.E. due to secondary}-\overbrace{\frac{Gm\pqty{M_1+M_2}\pqty{x^2+y^2}}{2a^3}}^\text{centrifugal potential}.
\end{equation}

We introduce the dimensionless coordinates measured in units of the separation of the binary (or reduced masses)

\begin{equation}
    \xi = \frac{x}{a},\quad\eta=\frac{y}{a},\quad\xi_1=\frac{M_2}{M_1+M_2},\quad\xi_2=\frac{M_1}{M_1+M_2}
\end{equation}

Our potential now becomes

\begin{equation}
    V(\xi,\eta) = -\frac{Gm\pqty{M_1+M_2}}{a}\times\bqty{\frac{\xi_2}{\sqrt{\pqty{\xi-\xi_1}^2+\eta^2}}+\frac{\xi_1}{\sqrt{\pqty{\xi-\xi_2}^2+\eta^2}}+\frac12\pqty{\xi^2+\eta^2}}.
\end{equation}

Taking the partial derivative along the $\xi$-axis (the rescaled $x$-axis), we get

\begin{equation}
    \pdv{V}{\xi} = -\frac{Gm\pqty{M_1+M_2}}{a}\times\bqty{-\frac{\xi_2\pqty{\xi-\xi_1}}{\pqty{\pqty{\xi-\xi_1}^2+\eta^2}^\frac32}-\frac{\xi_1\pqty{\xi-\xi_2}}{\pqty{\pqty{\xi-\xi_2}^2+\eta^2}^\frac32}+\xi}\label{eq:pdv_of_roche_potential}
\end{equation}

Hence the extrema of the potential $V$ are at the points $\xi$ where (taking $\eta=0$ on the $\xi$-axis)

% -ve sign on first two terms dumbass

\begin{tcolorbox}
    \begin{equation}
        0=\xi-\frac{\xi_2}{\pqty{\xi-\xi_1}^2}-\frac{\xi_1}{\pqty{\xi-\xi_2}^2},
    \end{equation}
\end{tcolorbox}

which was to be shown.

\section{Kepler's Three Laws}\label{chap:keplers_laws}

This section is adapted from \citet{Richards:2021}. Kepler's three laws such that they apply to binary stars are:

\begin{enumerate}
    \item All binary stars orbit the barycentre in an ellipse with the barycentre as a focus.
    \item A line segment joining the two stars sweeps out equal areas during equal intervals of time.
    \item The square of the period of each star is proportional to the cube of the semimajor axis (i.e., $T^2\propto a^3$).
\end{enumerate}

The derivations are presented below.

\subsection{The First Law}\label{sec:keplers_first_law}

\begin{figure}[!htbp]
    \centering
    \input{chapters/figures/binary_schematic}
    \caption[A schematic of a simple binary star system.]{\Cref{fig:bss_schematic} again: A schematic of a simple binary star system with eccentricity $e$, semimajor axis $a$, and semiminor axis $b$. The centre of mass (barycentre) is shown as a blue dot. Here, $M_1$ is the heavier star, which orbits at a smaller distance $r_1$ to the barycentre by Kepler's first and Newton's second laws. The elliptical orbital trajectory is shown in grey. Figure adapted from \citet{Richards:2021}.}
    \label{fig:bss_schematic_appendix}
\end{figure}

Given the orbital setup in \Cref{fig:bss_schematic_appendix}, the distance vector $\vec{r}$ between two stars in a binary is

\begin{equation}
    \vec{r} = \vec{r}_2 - \vec{r}_1,
\end{equation}

with the reduced mass

\begin{equation}
    \mu = \frac{M_1 M_2}{M_1+M_2},
\end{equation}

and total mass

\begin{equation}
    M = M_1 + M_2.
\end{equation}

Since the angular momentum vector is given by

\begin{equation}
    \vec{L} = \vec{r}\times\vec{p},
\end{equation}

its derivative is

\begin{equation}
    \dv{\vec{L}}{t} = \dv{\vec{r}}{t}\times\vec{p} + \vec{r}\times\dv{\vec{p}}{t} = 0,\label{eq:momentum_conserved}
\end{equation}

where the equality follows as $\vec{v}\equiv\dv{\vec{r}}{t}$ is parallel to $\vec{p}$, and $\vec{r}$ is parallel to $\dv{\vec{p}}{t}\equiv \vec{F}$. This is equivalent to a one-body problem where a mass $\mu=\frac{M_1 M_2}{M_1+M_2}$ orbits a central mass of mass $M_1+M_2$. The angular momentum is

\begin{align}
    \vec{L} &= \mu\vec{r}\times\vec{v} \label{eq:ang_mom_def}\\
            &= \mu \abs{\vec{r}}\vu{r}\times\dv{}{t}\pqty{\abs{\vec{r}}\vu{r}} \notag\\
            &= \mu \abs{\vec{r}}\vu{r}\times\pqty{\dv{r}{t}\vu{r}+r\dv{\vu{r}}{t}} \notag\\
            &= -\dv{r}{t}\vu{r}\times\mu \abs{\vec{r}}\vu{r} - r\dv{\vu{r}}{t}\times\mu \abs{\vec{r}}\vu{r} \notag\\
            &= -\dv{r}{t}\mu \abs{\vec{r}}\pqty{\vu{r}\times\vu{r}} - \abs{\vec{r}}^2\mu\pqty{\dv{\vu{r}}{t}\times\vu{r}} \notag\\
            &= -\abs{\vec{r}}^2\mu\pqty{\dv{\vu{r}}{t}\times\vu{r}} \notag\\
            &= \abs{\vec{r}}^2\mu\pqty{\vu{r}\times\dv{\vu{r}}{t}} \notag\\
            &= \mu \abs{\vec{r}}^2\vu{r}\times\dv{\vu{r}}{t}. \label{eq:bss_ang_mom}
\end{align}

We seek an expression for $\abs{\vec{r}}$ to determine the trajectory of the stars. We write

\begin{align*}
    \vec{a}\times\vec{L} &= -\frac{GM}{\abs{\vec{r}}^2}\vu{r}\times\pqty{\mu \abs{\vec{r}}^2\vu{r}\times\dv{\vu{r}}{t}} \\
    &= GM\mu\vu{r}\times\pqty{\vu{r}\times\dv{\vu{r}}{t}} \\
    &= -GM\mu\bqty{\pqty{\vu{r}\cdot\dv{\vu{r}}{t}}\vu{r}-\pqty{\vu{r}\cdot\vu{r}}\dv{\vu{r}}{t}} \\
    &= GM\mu\dv{\vu{r}}{t}.
\end{align*}

Now consider

\begin{align}
    \dv{}{t}\pqty{\vec{v}\times\vec{L}} &= \dv{\vec{v}}{t}\times\vec{L}+\vec{v}\times\dv{\vec{L}}{t} \notag\\
    &= \vec{a}\times\vec{L} \notag\\
    &= \dv{}{t} \left(GM\mu\vu{r}\right),
\end{align}

where $\dv{\vec{L}}{t}=0$ by \Cref{eq:momentum_conserved}. Hence

\begin{equation}
    \vec{v}\times\vec{L} = GM\mu\vu{r} + \vec{D},
\end{equation}

where $\vec{D}$ is a constant of integration that exists in the plane of the orbit. We can then introduce the vector $\vec{r}$ into this equation:

\begin{align}
    \vec{r}\cdot\pqty{\vec{v}\times\vec{L}} &= GM\mu \abs{\vec{r}}\vu{r}\cdot\vu{r}+\vec{r}\cdot\vec{D} \notag\\
    \pqty{\vec{r}\times\vec{v}}\cdot L &= GM\mu r + rD\cos\theta.
\end{align}

Now by \Cref{eq:bss_ang_mom}, this is simply

\begin{equation}
    \frac{\abs{\vec{L}}^2}{\mu} = GM\mu \abs{\vec{r}}\pqty{1 + \frac{\abs{\vec{D}}\cos\theta}{GM\mu}}.
\end{equation}

We define the eccentricity of the ellipse as

\begin{equation}
    e = \frac{\abs{\vec{r}}}{GM\mu}
\end{equation}

and hence we have

\begin{tcolorbox}[enhanced,title={Kepler's First Law}]
    \begin{equation}
        \abs{\vec{r}}=\frac{\abs{\vec{L}}^2 / \mu^2}{GM\pqty{1+e\cos\theta}},
        \label{eq:keplers_first_law}
    \end{equation}
\end{tcolorbox}

which is the equation defining an ellipse, provided that\footnote{Or equivalently that $\abs{\vec{D}}<GM\mu$, which we are free to select as required as the vector $\vec{D}$ is arbitrary.} $e<1$ -- thus proving the First Law. Since the general equation for an ellipse with semimajor axis $a$ is

\begin{equation}
    \abs{\vec{r}} = \frac{a\pqty{1-e^2}}{1+e\cos\theta}\quad\text{ for } 0 \leq e < 1,
\end{equation}

it is immediately apparent that we can write

\begin{equation}
    \abs{\vec{L}} = \mu\sqrt{GMa(1-e^2)},\label{eq:angular_momentum_elliptical_orbit}
\end{equation}

and since $\abs{\vec{L}}$ is constant by \Cref{eq:momentum_conserved}, presuming total mass $M$ remains constant we have that $\mu$ is constant as well.

Consider the area of an elliptical orbit (see the shaded area in \Cref{fig:secondlaw_sketch}). We have

\begin{equation}
    \dd{A} = \dd{r}\pqty{r\dd{\theta}} = r\dd{r}\dd{\theta},
\end{equation}

and integrating over $r$ gives

\begin{equation}
    \dd{A} = \frac12 r^2\dd{\theta},
\end{equation}

thus

\begin{equation}
    \dv{A}{t} = \frac12 r^2\dv{\theta}{t}.
\end{equation}

We can write the velocity of the star in the plane of the ellipse in terms of its radial component $\vec{v}_\text{r}$ and its angular component $\vec{v}_\theta$,

\begin{equation}
    \vec{v} = \vec{v}_\text{r} + \vec{v}_\theta = \dv{r}{t}\vu{r} + \abs{\vec{r}}\dv{\theta}{t}%\vu{\theta}.
\end{equation}

Importantly we have

\begin{equation}
    v_\theta = r\dv{\theta}{t},
\end{equation}

such that

\begin{equation}
    \dv{A}{t} = \frac12 r v_\theta.
\end{equation}

Since $r$ and $v_\theta$ are orthogonal, we have

\begin{equation}
    rv_\theta = \underbrace{\abs{\vec{r}\times\vec{\theta}} = \abs{\frac{\vec{L}}{\mu}}}_\text{by \Cref{eq:ang_mom_def}} = \frac{L}{\mu},
\end{equation}

hence

\begin{tcolorbox}[enhanced,title={Kepler's Second Law}]
    \begin{equation}
        \dv{A}{t} = \frac12 \frac{L}{\mu}.
        \label{eq:second_law}
    \end{equation}
\end{tcolorbox}

By \Cref{eq:momentum_conserved}, $L$ is constant (and by \Cref{eq:angular_momentum_elliptical_orbit}, $\mu$ is constant as well), hence we have that $\dv{A}{t}$ is constant over the orbit, and Kepler's second law is established.

\subsection{The Third Law}

Let the period of the binary star system be given by $T$. The area of the ellipse swept out by the binary over its orbit is

\begin{equation}
    A = \int_0^T \dv{A}{t} \dd{t} = \frac12 \frac{L}{\mu}\int_0^T \dd{t} = \frac12 \frac{L}{\mu} T,
\end{equation}

and since the orbit is elliptical, this is equal to $\pi ab$, and thus

\begin{equation}
    \pi ab = \frac12 \frac{L}{\mu} T,
\end{equation}

or, equivalently,

\begin{equation}
    T^2 = \frac{4\pi^2a^2b^2\mu^2}{L^2}.
\end{equation}

Further, from the geometry of an ellipse we have

\begin{equation}
    b^2 = a^2\pqty{1-e^2}.
\end{equation}

Using this and \Cref{eq:angular_momentum_elliptical_orbit} in tandem, we have

\begin{tcolorbox}[enhanced,title={Kepler's Third Law}]
    \begin{equation}
        T^2 = \frac{4\pi^2}{GM} a^3,
    \end{equation}
\end{tcolorbox}

which proves Kepler's third law.

\section{Equations of Motion of decaying binaries}

The following derivation is adapted from the derivation presented in \citet{Richards:2021}. In this section we use a positive-definite metric signature ($\tr{\tens{\eta}}=+2$), and the Einstein summation convention where repeated indices are summed shall apply. Further, Greek indices span from 0 to 3, and Latin indices span from 1 to 3.

We have two goals. The first is to derive expressions for the coupled equations $\dv{a}{t}$ and $\dv{e}{t}$ describing the decay of a binary due to gravitational radiation. The second is to determine an expression for $T_\text{c}(a_0, e_0)$, the coalescence time, which is the time-scale for this to occur.

\subsection{\texorpdfstring{$\dv{a}{t}$ and $\dv{e}{t}$}{Rate of change of semimajor axis and eccentricity}}

In the weak-field limit, the spacetime metric is presumed to be a perturbation of a locally Minkowski metric, that is, it is given by

\begin{equation}
    g_{\mu\nu} = \eta_{\mu\nu} + \kappa h_{\mu\nu},
\end{equation}

with \tens{\eta} the Minkowski metric, $\abs{h}\ll 1$, and $\kappa^2 = 32\pi G$. This metric induces \citep[see, among others,][]{Einstein:1916, Peters:1964} the wave equation

\begin{equation}
    \Box \overline{h}_{\mu\nu} = -\frac12 \kappa T_{\mu\nu},
\end{equation}

where \tens{T} is the stress-energy tensor. Given some plane wave solution to this wave equation, of the form

\begin{equation}
    h_{\mu\nu}=Ae_{\mu\nu}\cos\pqty{\vec{k}\cdot\vec{x}-\omega t},
\end{equation}

where \tens{e} is a unit polarization tensor, we have the energy density

\begin{equation}
    U = \frac12 c^2\omega^2A^2.
\end{equation}

The two possible non-zero spacelike and transverse polarizations for a wave travelling (without loss of generality) in the $z$-direction are

\begin{equation}
    e_1 = \frac{1}{\sqrt{2}}\pqty{\hat{x}\hat{x}-\hat{y}\hat{y}},\\
\end{equation}

and

\begin{equation}
    e_2 = \frac{1}{\sqrt{2}}\pqty{\hat{x}\hat{y}-\hat{y}\hat{x}},
\end{equation}

where we denote the ``plus'' polarisation as $e_1$ and the ``cross'' polarisation as $e_2$. The power $P$ radiated into a solid angle $\Omega$ is given by

\begin{equation}
    \dv{P}{\Omega} = \frac{G}{8\pi c^5}\pqty{\pdv[3]{Q_{ij}}{t}e_{ij}}^2,
\end{equation}

where the components of the mass tensor \tens{Q} are given by

\begin{equation}
    Q_{ij} = m^\alpha x_{\alpha i}x_{\alpha j}.
\end{equation}

The total power is obtained reasonably straightforwardly:

\begin{equation}
    \dv{P}{\Omega} = \frac{G}{8\pi c^5} \left[\pqty{\dv[3]{Q_{ij}}{t}}^2-2n_i \dv[3]{Q_{ij}}{t}n_k\dv[3]{Q_{kj}}{t}-\frac12\pqty{\dv[3]{Q_{ii}}{t}}^2+\frac12\pqty{n_i n_j\dv[3]{Q_{ij}}{t}}^2+\dv[3]{Q_{ii}}{t}n_j n_k\dv[3]{Q_{jk}}{t}\right],
\end{equation}

with $\hat{n}$ a unit vector in the direction of radiation. The total power is then found by integrating this equation over all directions, viz.

\begin{equation}
    P = \frac{G}{5c^5}\pqty{\dv[3]{Q_{ij}}{t}\dv[3]{Q_{ij}}{t}-\frac13\dv[3]{Q_{ii}}{t}\dv[3]{Q_{jj}}{t}}.
\end{equation}

Let us now inject the physics. Suppose the primary star has mass $M_1$ and the secondary mass $M_2$. In the $xy$-plane, with the origin at the center of mass, we can parameterise the positions of the two stars as

\begin{equation}
    \vec{x}_1 = \begin{bmatrix}d_1\cos\varphi \\ d_1\sin\varphi\end{bmatrix} \qquad \vec{x}_2 = \begin{bmatrix}-d_2\cos\varphi \\ -d_2\sin\varphi\end{bmatrix},
\end{equation}

with

\begin{equation}
    d_1 = \frac{m}{m+M}d\quad\text{and}\quad d_2 = \frac{M}{m+M}d,
\end{equation}

and $\varphi$ our angle parameter. The components of the tensor \tens{Q} are

\begin{align}
    Q_{xx} &= \mu d^2\cos^2\varphi, \\
    Q_{yy} &= \mu d^2\sin^2\varphi, \\
    Q_{xy} = Q_{yx} &= \mu d^2\sin\varphi\cos\varphi,
\end{align}

with the rest vanishing, and with $\mu$ as the reduced mass. Now, as binary stars obey Kepler's laws, Kepler's orbit equation (\Cref{eq:keplers_first_law}) holds:

\begin{equation}
    d = \frac{a(1-e^2)}{1+e\cos\dot{\varphi}},
\end{equation}

whilst the angular velocity is given by

\begin{equation}
    \dot{\varphi} = \frac{\sqrt{G(M_1+M_2)(1-e^2)a}}{d^2}.
\end{equation}

We therefore have

\begin{align}
    \dv[3]{Q_{xx}}{t} &= \gamma (1+e\cos\dot{\varphi})^2(2\sin(2\dot{\varphi})+3e\sin\dot{\varphi}\cos^2\dot{\varphi}), \\
    \dv[3]{Q_{yy}}{t} &= -\gamma(1+e\cos\dot{\varphi})^2\pqty{2\sin(2\dot{\varphi})+e\sin\dot{\varphi}(1+3\cos^2\dot{\varphi}}, \\
    \dv[3]{Q_{xy}}{t} = \dv[3]{Q_{yx}}{t} &= -\gamma(1+e\cos\dot{\varphi})^2\pqty{2\cos(2\dot{\varphi})-e\cos\dot{\varphi}(1-3\cos^2\dot{\varphi})},
\end{align}

where the quantity $\gamma$ is defined as

\begin{equation}
    \gamma^2 \equiv \frac{4G^3M_1^2M_2^2(M_1+M_2)}{a^5(1-e^2)^5}.
\end{equation}

The total power is therefore

\begin{equation}
    P = \frac{8}{15}\frac{G^4}{c^5}\frac{M_1^2M_2^2(M_1+M_2)}{a(1-e^2)^5}(1+e\cos\varphi)^4\pqty{12(1+e\cos\varphi)^2+e^2\sin^2\varphi}.
\end{equation}

Averaging this over one orbit yields the average power, which we write now as $\ev{\dv{E}{t}}$,

\begin{equation}
    \ev{\dv{E}{t}} = -\frac{32}{5}\frac{G^4}{c^5}\frac{M_1^2M_2^2(M_1+M_2)}{a^5(1-e^2)^\frac72}\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4}.
    \label{eq:average_power}
\end{equation}

The average angular momentum is

\begin{equation}
    \ev{\dv{L}{t}} = -\frac{32}{5}\frac{G^\frac72 M_1^2M_2^2\sqrt{M_1+M_2}}{c^5 a^\frac72 \pqty{1-e^2}^2}\pqty{1+\frac78 e^2}.
    \label{eq:average_ang_mom}
\end{equation}

We now write the gravitational energy from the two-body problem,

\begin{align}
    E &= -\frac{GM_1 M_2}{2a} \\
    \implies \dv{E}{t} &= \frac{GM_1 M_2}{2}\frac{1}{a^2}\dv{a}{t}.\label{eq:average_power_2body}
\end{align}

Define the constant $\beta$ as

\begin{equation}
    \beta \equiv \frac{64}{5}\frac{G^3M_1 M_2\pqty{M_1+M_2}}{c^5}.
\end{equation}

Now insert \Cref{eq:average_power_2body} into \Cref{eq:average_power} to obtain

\begin{align}
    \dv{a}{t} &= -\frac{2a^2}{GM_1 M_2}\frac{32}{5}\frac{G^4M_1^2M_2^2\pqty{M_1+M_2}}{c^5a^5\pqty{1-e^2}^\frac72}\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4} \\
    &= -\frac{\beta}{a^3\pqty{1-e^2}^\frac72}\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4},
\end{align}

which is our first equation. The derivation for $\dv{e}{t}$ is more involved. The angular momentum from the two-body problem is

\begin{equation}
    L^2 = \frac{GM_1^2M_2^2a\pqty{1-e^2}}{M_1+M_2}
\end{equation}

Implicit differentiation reveals that

\begin{align}
    2L\dv{L}{t} &= \frac{GM_1^2M_2^2}{M_1+M_2}\dv{}{t} \pqty{a\pqty{1-e^2}} \\
    &= \frac{GM_1^2M_2^2}{M_1+M_2}\bqty{\dv{a}{t}\pqty{1-e^2}-2ae\dv{e}{t}}.
\end{align}

Multiplying each side of \Cref{eq:average_ang_mom} by $2L$ and substituting this in, we obtain

\begin{align}
    \frac{GM_1^2M_2^2}{M_1+M_2}\bqty{\dv{a}{t}\pqty{1-e^2}-2ae\dv{e}{t}} &= -L\frac{64}{5}\frac{G^\frac72 M_1^2M_2^2\sqrt{M_1+M_2}}{c^5 a^\frac72 \pqty{1-e^2}^2}\pqty{1+\frac78 e^2} \notag\\
    \frac{1}{M_1+M_2}\bqty{\dv{a}{t}\pqty{1-e^2}-2ae\dv{e}{t}} &= -\sqrt{\frac{GM_2^2M_1^2a\pqty{1-e^2}}{M_1+M_2}}\frac{64}{5}\frac{G^\frac52 \sqrt{M_1+M_2}}{c^5 a^\frac72 \pqty{1-e^2}^2}\pqty{1+\frac78 e^2} \notag\\
    \dv{a}{t}\pqty{1-e^2}-2ae\dv{e}{t} &= -\frac{\beta}{a^3 \pqty{1-e^2}^\frac32}\pqty{1+\frac78 e^2} \notag\\
    2ae\dv{e}{t} &= \frac{\beta}{a^3 \pqty{1-e^2}^\frac32}\pqty{1+\frac78 e^2}+\dv{a}{t}\pqty{1-e^2} \notag\\
    2ae\dv{e}{t} &= \frac{\beta}{a^3 \pqty{1-e^2}^\frac32}\pqty{1+\frac78 e^2}-\frac{\beta}{a^3\pqty{1-e^2}^\frac52}\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4} \notag\\
    \dv{e}{t} &= \frac{\beta}{a^4 \pqty{1-e^2}^\frac52}\frac{\pqty{1-e^2}\pqty{1+\frac78 e^2}-\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4}}{2e} \notag\\
    &= \frac{\beta}{a^4 \pqty{1-e^2}^\frac52}\bqty{-\frac{19}{12}e^3-\frac{121}{192} e} \notag\\
    &= -\frac{19}{12}\frac{\beta}{a^4 \pqty{1-e^2}^\frac52}\pqty{e+\frac{121}{304}e^3},
\end{align}

which is our second equation. Hence we arrive at the coupled differential equations describing the decay of a binary star system:

\begin{tcolorbox}
    \begin{subequations}
        \begin{align}
            \dv{a}{t} &= \frac{-\beta}{a^3\pqty{1-e^2}^\frac72}\pqty{1+\frac{73}{24}e^2+\frac{37}{96}e^4}\label{eq:peters_sma} \\
            \dv{e}{t} &= -\frac{19}{12}\frac{\beta}{a^4\pqty{1-e^2}^\frac52}\pqty{e+\frac{121}{304}e^3}\label{eq:peters_ecc}
        \end{align}
        \label{eq:peters_coupled_equations}
    \end{subequations}
\end{tcolorbox}

\subsection{The coalescence time}

The coalescence time is by definition the time it takes for the binary system to merge after the secondary supernova. We can step towards this in a similar way to \citet{Peters:1964}. First, dividing \Cref{eq:peters_sma} by \Cref{eq:peters_ecc} gives

\begin{equation}
    \dv{a}{e} = \frac{12}{19}\frac{a}{e}\frac{1+\frac{73}{24}e^2+\frac{37}{96}e^4}{\pqty{1-e^2}\pqty{1+\frac{121}{304}e^2}}
\end{equation}

We can integrate this however we choose to obtain $a(e)$, namely

\begin{equation}
    a(e) = \frac{c_0 e^\frac{12}{19}}{\pqty{1-e^2}}\pqty{1+\frac{121}{304}e^2}^\frac{870}{2299}
    \label{eq:a_functionof_e}
\end{equation}

with

\begin{equation}
    c_0 \equiv a_0\pqty{1-e_0^2}e_0^{-\frac{12}{19}}\pqty{1+\frac{121}{304}e_0^2}^{-\frac{870}{2299}}
    \label{eq:singularity_in_c0_oh_noes}
\end{equation}

embodying the initial parameters. Now, substituting \Cref{eq:a_functionof_e} into \Cref{eq:peters_ecc}, we see

\begin{align}
    \dv{e}{t} &= -\frac{19}{12}\frac{\beta}{a^4\pqty{1-e^2}^\frac52}\pqty{e+\frac{121}{304}e^3} \\
    &= -\frac{19}{12}\frac{\beta}{\pqty{\frac{c_0 e^\frac{12}{19}}{\pqty{1-e^2}}\pqty{1+\frac{121}{304}e^2}^\frac{870}{2299}}^4\pqty{1-e^2}^\frac52}\pqty{e+\frac{121}{304}e^3} \\
    &= -\frac{19}{12}\frac{\beta}{c_0^4}\frac{\pqty{1-e^2}^\frac{3}{2} e}{e^\frac{48}{19}}\frac{\pqty{1+\frac{121}{304}e^2}}{\pqty{1+\frac{121}{304}e^2}^\frac{3480}{2299}} \\
    &= -\frac{19}{12}\frac{\beta}{c_0^4}\frac{\pqty{1-e^2}^\frac{3}{2} e^{-\frac{29}{19}}}{\pqty{1+\frac{121}{304}e^2}^\frac{1181}{2299}} \\
\end{align}

Now write this as $\dv{t}{e}$ and integrate to get the coalescence time:

\begin{align}
    \dv{t}{e} &= -\frac{12}{19}\frac{c_0^4}{\beta}\frac{\pqty{1+\frac{121}{304}e^2}^\frac{1181}{2299}}{\pqty{1-e^2}^\frac{3}{2} e^{-\frac{29}{19}}} \\
    T_\text{c} = \int_0^{T_\text{c}} \dd{t} &= -\frac{12}{19}\frac{c_0^4}{\beta}\int_{e_0}^0\frac{\pqty{1+\frac{121}{304}e^2}^\frac{1181}{2299}}{\pqty{1-e^2}^\frac{3}{2} e^{-\frac{29}{19}}} \dd{e} \\
    &= \frac{12}{19}\frac{c_0^4}{\beta}\int_0^{e_0}\frac{\pqty{1+\frac{121}{304}e^2}^\frac{1181}{2299}}{\pqty{1-e^2}^\frac{3}{2} e^{-\frac{29}{19}}} \dd{e}
\end{align}

i.e.,

\begin{tcolorbox}
    \begin{equation}
        T_\text{c}(a_0, e_0) = \frac{12}{19}\frac{c_0^4}{\beta}\int_0^{e_0} \frac{e^\frac{22}{19}\pqty{1+\frac{121}{304}e^2}^\frac{1181}{2299}}{\pqty{1-e^2}^\frac32}\dd{e}
        \label{eq:coalescence_time_formal}
    \end{equation}
\end{tcolorbox}

If $e_0=0$ then the orbit is circular, and we immediately see concerning behaviour -- the integral collapses to zero, necessitating a different approach.

One of the critical mistakes here is presuming that $c_0$ in \Cref{eq:singularity_in_c0_oh_noes} is valid for all physical eccentricities. In fact, in the limit as $e_0 \to 0$, $c_0$ diverges. We retreat to before this critical assumption, and examine \Cref{eq:peters_sma}. In the limiting case of a circular orbit, $e=e_0=0$, so the expression becomes

\begin{equation}
    \dv{a}{t} = -\frac{\beta}{a^3}.
\end{equation}

and by separation of variables we obtain

\begin{equation}
    a^3 \dd{a} = -\beta\dd{t},
\end{equation}

of which the solution is

\begin{equation}
    \frac{a^4}{4} = -\beta t + c,
\end{equation}

for constant $c\in\mathbb{R}$. The boundary conditions are that when $t=0, a=a_0$, and when $t=T_\text{c}, a=0$. Applying the first boundary condition, we have $c=a_0^4$. Hence,

\begin{equation}
    a(t) = \pqty{a_0^4-4\beta t}^\frac14 ,
\end{equation}

and imposing our second boundary condition we obtain the coalescence time

\begin{equation}
    T_\text{c} = \frac{a_0^4}{4\beta}.
\end{equation}

But if $e\neq 0$, then the orbit is eccentric and the integral must be solved numerically, as no closed form solution exists.

\section{The Chandrasekhar mass}

The following derivation is adapted from the one in \citet{Eldridge:2018}.

\subsection{The Lane-Emden equation}

We begin from the equations of stellar structure, \Cref{eq:stellar_structure}. We take as our equation of state pressure to be to a power of density, i.e.,

\begin{equation}
    P\propto\rho^\gamma \implies P=K\rho^{1+\frac{1}{n}}
    \label{eq:polytrope_defn}
\end{equation}

with $K$ a constant and $n$ (defined by $\gamma=1+\frac{1}{n}$) the so-called polytropic index. We use \Cref{eq:stellar_structure} to establish an ordinary differential equation for the pressure $P$.

\begin{align}
    \dv{P}{r} &= -\frac{Gm\rho}{r^2} \\
    \frac{r^2}{\rho}\dv{P}{r} &= -Gm \\
    \dv{}{r}\pqty{\frac{r^2}{\rho}\dv{P}{r}} &= -G\dv{m}{r} = -4\pi G\rho r^2 \\
    \frac{1}{r^2}\dv{}{r}\pqty{\frac{r^2}{\rho}\dv{P}{r}} &= -4\pi G\rho.
    \label{eq:yikes_ode}
\end{align}

To solve this differential equation, we introduce a dimensionless parameter $\theta$ defined by

\begin{equation}
    \rho = \rho_\text{c}\theta^n,
\end{equation}

with $\theta=1$ at $r=0$ and $\theta=0$ at $r=R$. The first constraint defines $\rho_\text{c}$ as the central density. Using this in \Cref{eq:polytrope_defn} we have

\begin{align}
    P &= K\pqty{\rho_\text{c}\theta^n}^{1+\frac{1}{n}} \\
    &= K\rho_\text{c}^{1+\frac{1}{n}}\theta^{n\pqty{1+\frac{1}{n}}} \\
    &= P_\text{c}\theta^{n+1}
\end{align}

again defining $P_\text{c}$ as the central pressure. Substituting this into \Cref{eq:yikes_ode} we have

\begin{align}
    \frac{1}{r^2}\dv{}{r}\pqty{\frac{r^2}{\rho}\bqty{K\rho_\text{c}^{1+\frac{1}{n}}\pqty{1+n}\theta^n\dv{\theta}{r}}} &= -4\pi G\rho \\
    \frac{(n+1)K}{4\pi G}\frac{1}{r^2}\dv{}{r}\pqty{\frac{r^2}{\rho}\rho_\text{c}^{1+\frac{1}{n}}\theta^n\dv{\theta}{r}} &= -\rho_\text{c} \theta^n \\
    \frac{(n+1)K}{4\pi G}\rho_\text{c}^{\frac{1}{n}}\frac{1}{r^2}\dv{}{r}\pqty{\frac{r^2}{\rho_\text{c}\theta^n}\theta^n\dv{\theta}{r}} &= -\theta^n \\
    \frac{(n+1)K}{4\pi G}\rho_\text{c}^{\frac{1}{n}-1}\frac{1}{r^2}\dv{}{r}\pqty{r^2\dv{\theta}{r}} &= -\theta^n
\end{align}

Now we introduce the parameter $\zeta$ by defining

\begin{equation}
    r=\alpha\zeta \qquad\text{where}\qquad \alpha^2\equiv\frac{(n+1)K}{4\pi G}\rho_\text{c}^{\frac{1}{n}-1} \label{eq:r_def}
\end{equation}

therefore $\dd{r}=\alpha\dd{\zeta}$ and we have

\begin{tcolorbox}
    \begin{equation}
        \frac{1}{\zeta^2}\dv{}{\zeta}\pqty{\zeta^2\dv{\theta}{\zeta}} = -\theta^n
    \end{equation}
\end{tcolorbox}

which is known as the Lane-Emden Equation.

\subsection{Mass and Radius relation}

From the Lane-Emden equation, one can deduce the dimensionless radius $R$ of the star for a given polytropic index $n$ via $R=\alpha X_n$, where $X_n$ is the zero of $\theta$ in $\zeta$. From this definition, we have

\begin{align}
    R &= \alpha X_n \\
    &\propto \sqrt{K\rho_\text{c}^{\frac{1}{n}-1}},
\end{align}

the proportionality constant being $\sqrt{\frac{n+1}{4\pi G}}$. Constructing the mass is similar, where the mass internal to radius $r$ is

\begin{align}
    m(r) &= \int_0^r 4\pi \rho r^\prime \dd{r^\prime} \\
    &= 4\pi \rho_\text{c} \alpha^3 \int_0^\zeta  \theta^n \pqty{\zeta^\prime}^2 \dd{\zeta^\prime}.
\end{align}

The second step arises from taking $r=\alpha \zeta$ and changing the bounds of integration. Substituting in the Lane-Emden equation we obtain

\begin{align}
    m(r) &= -4\pi\rho_\text{c} \alpha^3 \int_0^\zeta \dv{}{{\zeta^\prime}} \pqty{\bqty{\zeta^\prime}^2 \dv{\theta}{{\zeta^\prime}}}\dd{\zeta^\prime} \\
    &= -4\pi \rho_\text{c} \alpha^3 \zeta^2 \dv{\theta}{\zeta}.
\end{align}

Hence the total mass is

\begin{align}
    M = 4\pi \pqty{\frac{\pqty{n+1}K}{4\pi G}}^\frac32 \rho_\text{c}^\frac{3-n}{2n} \pqty{-\zeta^2 \dv{\theta}{\zeta}}_{\zeta=X_n}.\label{eq:mass_relationship_homology}
\end{align}

Or, as a proportionality relationship,

\begin{equation}
    M \propto K^\frac32 \rho_\text{c}^\frac{3-n}{2n}. \label{eq:M_propto_rho}
\end{equation}

The relationship between $M$ and $R$ is thus

\begin{equation}
    R \propto M^\frac{n-1}{n-3},
\end{equation}

for fixed $K$. There is a singularity at $n=3$, depicted in \Cref{fig:polytropic_radii} for stars in the range \SI{5}{\msun} to \SI{55}{\msun}. An $n=3$ polytrope corresponds to the equation of state $P\propto\rho^\frac43$, and by \Cref{eq:M_propto_rho}, has its mass independent of the radius or the density.

    \caption{The radius of stars of varying mass as a function of the polytropic index they are modelled at.}
    \label{fig:polytropic_radii}
\end{figure}

\subsection{The limiting mass}

The number density of electrons in a white dwarf is given by

\begin{equation}
    n = \frac{N}{V} = \frac{\rho}{\mu m_\text{H}},
\end{equation}

where $\mu$ is the mean molecular weight of an electron ($\mu=2$ in a stellar core) and $m_\text{H}$ is the mass of a hydrogen atom.

A calculation of the degeneracy pressure of ultrarelativistic electrons in this configuration is beyond the scope of this thesis, but can be written as

\begin{equation}
    P_\text{degen} = \frac{c\hbar}{12\pi^2}\pqty{\frac{3\pi^2 N}{V}}^\frac43,
\end{equation}

and thus

\begin{equation}
    P_\text{degen} = \frac{c\hbar}{12\pi^2}\pqty{\frac{3\pi^2}{2m_\text{H}}}^\frac43 \rho^\frac43.
\end{equation}

This is a polytrope with $n=3$ and

\begin{equation}
    K = \frac{c\hbar}{12\pi^2}\pqty{\frac{3\pi^2}{2m_\text{H}}}^\frac43 = \SI{1.224e15}{\dyne\per\centi\meter\squared(\gram\per\centi\meter\cubed)^{-4/3}}
\end{equation}

As before, this material cannot remain bound to the star -- the radius tends to infinity at $n=3$ per \Cref{fig:polytropic_radii} -- so we have the maximum mass of a stable star supported by electron degeneracy pressure as

\begin{equation}
    M = 2.02\cdot4\pi\pqty{\frac{K}{\pi G}} \approx \frac{5.75}{\mu^2} \unit{\msun},
\end{equation}

where the coefficient of 2.02 is evaluated numerically from the $\pqty{-\zeta^2 \dv{\theta}{\zeta}}_{\zeta=X_n}$ term in \Cref{eq:mass_relationship_homology}.

\begin{equation}
    M = M_\text{ch} \equiv \SI{1.438}{\msun}.
\end{equation}

Which was to be shown.

\section{Construction of the wind mass-loss rate}\label{chap:mass_loss_construction}

We construct our wind mass loss rates as follows. The total pre-WR mass loss is given by \citet{De_jager:1988}, namely

\begin{align}
    \log\dot{M}_\text{de Jager} &= -\sum_{n=0}^6 \sum_{\substack{i=0\\j=n-i}}^n a_{ij}\cos\left(i\arccos x_1\right)\cos\left(j\arccos x_2\right),\label{eq:chebyshev}\\
    \dot{M}_\text{pre-WR} &= \sqrt{\frac{Z}{Z_\odot}}10^{\log\dot{M}_\text{de Jager}},\label{eq:pre-wr-ml}
\end{align}

where $x_1 = \frac{\log T-4.05}{0.75}$, $x_2=\frac{\log L-4.6}{2.1}$, and the coefficients $a_{ij}$ are given by the entries in the matrix

\begin{equation}
    \mat{a} = \begin{pmatrix}
      6.34916 &  -5.04240 &  -0.83426 &  -1.13925 & -0.12202 \\
      3.41678 &   0.15629 &   2.96244 &   0.33659 &  0.57576 \\
     -1.08683 &   0.41952 &  -1.37272 &  -1.07493 &        - \\
      0.13095 &  -0.09825 &   0.13025 &         - &        - \\
      0.22427 &   0.46591 &         - &         - &        - \\
      0.11968 &         - &         - &         - &        - \\
    \end{pmatrix}.
\end{equation}

This mass loss rate is based upon the rates for 271 stars, the majority of which are Population I. Individual terms in the summation are Chebyshev polynomials of the first kind, restricted over the domain $\log~T_\text{eff}\in\left[3.3, 4.8\right]$, $\log~L\in\left[2.5, 6.7\right]$. As the free parameters here are temperature and luminosity, the result can be plotted over a \gls{hrd}. The results of this are shown in \Cref{fig:hrd_chebyshev}. Generally we constrain the inputs to the arccosine function to be between $-1 \leq x \leq 1$, which permits them to be defined outside of the defined area from \citet{De_jager:1988}; the effect of this is shown in the right-hand panel of \Cref{fig:hrd_chebyshev}. Without this restriction, the mass loss rates are simply undefined there, and are thus set to zero within the code.

The ``bounded parameters'' plot restricts the arguments to $\arccos$ in \Cref{eq:chebyshev} to be between $-1\leq x_i\leq 1$.}

We interpolate the bi-stability jump around \SI{25}{\kilo\kelvin} \citep[see, for example,][]{Pauldrach:1990, Vink:1999} with the following regime. Define the constants

\begin{align}
    C_1 &= \left(\frac{Z}{Z_\odot}\right)^{0.13},\label{eq:R1}\\
    C_2 &= -c_1+c_2\log\left(\frac{L}{c_{16}}\right)-c_3\log\left(\frac{M}{c_{15}}\right)-c_4\log\left(1.3C_1\right)+c_5\log\left(\frac{T}{c_{14}}\right)-c_6\left(\log\left[\frac{T}{c_{14}}\right]\right)^2+c_7\log\left(\frac{Z}{Z_\odot}\right),\label{eq:R2} \\
    &\text{and} \notag\\
    C_3 &= -c_8+c_9\log\left(\frac{L}{c_{16}}\right)-c_{10}\log\left(\frac{M}{c_{15}}\right)-c_{11}\log\left(0.65C_1\right)+c_{12}\log\left(\frac{T}{2\times 10^4}\right)+c_{13}\log\left(\frac{Z}{Z_\odot}\right),\label{eq:R3}
\end{align}

where the constants $c_i$ are given in \Cref{tab:bistability_constants}.

Our mass loss rate becomes:

\begin{equation}
    \dot{M}_\text{BS} = \begin{cases}
        \frac{(T-T_1)10^{C_3}+(T_2-T)\dot{M}_\text{pre-WR}}{2.5\times 10^3} & T_1 < T < T_2 \\
        10^{C_2} & T_2 \leq T \leq T_3 \\
        \frac{(T-T_3)10^{C_2}+(T_4-T)10^{C_3})}{5 \times 10^3} & T_3 < T < T_4 \\
        \frac{(6T_1-T)10^{C_2}+(T-5T_1)\dot{M}_\text{pre-WR}}{10^4} & 5T_1 < T < 6T_1 \\
        \dot{M}_\text{pre-WR} & \text{otherwise}
    \end{cases}
\end{equation}

where $T_1=\SI{10}{\kilo\kelvin}, T_2=\SI{12.5}{\kilo\kelvin}, T_3=\SI{22.5}{\kilo\kelvin}, T_4=\SI{27.5}{\kilo\kelvin}$. This interpolation is shown in \Cref{fig:bistability}. We define

\begin{equation}
    \mathcal{M} = \sqrt{\frac{Z}{Z_\odot}}\times 10^{-13.6 + 1.63\log\left(\frac{L}{L_\odot}\right)+2.22\log X},
\end{equation}

where $X$ is the surface hydrogen abundance of the primary, and then define

\begin{equation}
    \dot{M}_\text{WNL} = \begin{cases}
        5\left(\mathcal{M}-\dot{M}_\text{BS}\right)\left(0.42-X\right)+\dot{M}_\text{BS} & 0.42 > X > 0.4 \\
        \mathcal{M} & \text{otherwise}
    \end{cases}.
\end{equation}

Our total mass loss (provided the star has not depleted its surface helium, which will be discussed below) is now

\begin{equation}
    \dot{M}_\text{A} = \begin{cases}
        10\dot{\mu}\left(\log T-3.9\right) + \dot{M}_\text{BS} & X < 0.42 \text{ and } \log T > 3.9 \\
        \dot{M}_\text{WNL} & X > 0.42 \text{ and } \log T > 4.0 \\
        \dot{M}_\text{BS} & \text{otherwise}
    \end{cases},
\end{equation}

where $\dot{\mu}=\dot{M}_\text{WNL}-\dot{M}_\text{BS}$. Finally, if the surface helium depletes we modify the mass loss rate further, defining

\begin{equation}
    \dot{M}_\text{WC} = \sqrt{\frac{Z}{Z_\odot}}\times 10^{-8.3 + 0.84\log\left(\frac{L}{L_\odot}\right) + 2.04\log X_\text{He} + 1.04\log\left(1-X_\text{He}\right)}
\end{equation}

such that our total mass loss rate becomes (provided $X_\text{H} < 10^{-3}$ and $\log T > 4$)

\begin{equation}
    \dot{M}_\text{B} = \begin{cases}
        100\left(\dot{M}_\text{WC} - \dot{M}_\text{WNL}\right)\left(\mathcal{X}-0.02\right)+\dot{M}_\text{WNL} & \mathcal{X} > 0.02 \\
        \dot{M}_\text{WNL} & \mathcal{X} < 0.02
    \end{cases},
\end{equation}

where $\mathcal{X}=\frac{\frac13 X_\text{C}+\frac14 X_\text{O}}{X_\text{He}}$. The total wind mass-loss rate is then

\begin{equation}
    \dot{M}_\text{wind} = \begin{cases}
        \dot{M}_\text{B} & \text{surface helium depleted} \\
        \dot{M}_\text{A} & \text{otherwise}
    \end{cases}.\label{eq:vink_ml}
\end{equation}
